1. Offline Cash Acceptor (ESP32 + Coin / Bill Acceptor)

photobooth4u supports offline cash payment using any standard pulse-output coin slot (CH-926, JY-923, Allan) or bill validator (TB74, ICT, Apex) wired to an ESP32. No internet is needed — credit is tracked locally on the tablet or laptop and automatically starts a session once the price is reached.

Message protocol

Whenever a coin or bill finishes pulsing, the ESP32 sends one short line:

c<pesos>\n

Examples: c5\n adds ₱5, c20\n adds ₱20, c100\n adds ₱100. Line endings accepted: Enter / \n, \r, or ;. Valid single-note range: ₱1–₱5,000. Excess credit automatically rolls over to the next guest.

Wiring & 3.3V safety

  • Power your coin/bill acceptor with its own 12V DC power supply (do not draw 12V or acceptor current from the ESP32).
  • Tie 12V Supply GND and ESP32 GND together (common ground).
  • Wire Coin signal to GPIO 27 and Bill signal to GPIO 26 (configured with INPUT_PULLUP for standard open-collector / Normally Open acceptors).
  • Important: If your acceptor's signal line outputs active 5V or 12V, use a PC817 optocoupler (recommended) or a resistor voltage divider so the ESP32 GPIO never receives more than 3.3V.
  • Press the ESP32's built-in BOOT button (GPIO 0) any time to send a test c10\n credit.

Android tablet tip (BLE Keyboard mode)

When an ESP32 is paired as a Bluetooth keyboard, Android hides the virtual on-screen keyboard by default. Turn on Android Settings → System → Languages & input → Physical keyboard → Show on-screen keyboard so you can still type freely in Setup while the ESP32 stays connected.

ESP32 Arduino Sketch (esp32-cash-acceptor.ino)

// photobooth4u ESP32 Cash Acceptor (Coin + Bill -> Bluetooth / USB)
// BT_MODE: 1 = BLE HID Keyboard (works everywhere, incl. Android Offline App)
//          2 = BLE Nordic UART (Chrome/Edge, built-in ESP32 BLE library)
//          0 = USB Serial only (115200 baud is always on)
#define BT_MODE 1
#define COIN_PIN 27
#define BILL_PIN 26
#define TEST_BTN_PIN 0
#define LED_PIN 2
#define COIN_PESOS_PER_PULSE 1
#define BILL_PESOS_PER_PULSE 10
#define PULSE_DEBOUNCE_MS 22
#define COIN_TRAIN_GAP_MS 320
#define BILL_TRAIN_GAP_MS 420
#define DEVICE_NAME "Photobooth4u-Cash"

#if BT_MODE == 1
  #include <BleKeyboard.h>
  BleKeyboard bleKeyboard(DEVICE_NAME, "photobooth4u", 100);
#elif BT_MODE == 2
  #include <BLEDevice.h>
  #include <BLEServer.h>
  #include <BLEUtils.h>
  #include <BLE2902.h>
  #define NUS_SERVICE_UUID "6e400001-b5a3-f393-e0a9-e50e24dcca9e"
  #define NUS_TX_UUID      "6e400003-b5a3-f393-e0a9-e50e24dcca9e"
  BLECharacteristic *pTxChar = nullptr;
  bool bleClientConnected = false;
  class NusCallbacks : public BLEServerCallbacks {
    void onConnect(BLEServer *s) override { bleClientConnected = true; }
    void onDisconnect(BLEServer *s) override { bleClientConnected = false; BLEDevice::startAdvertising(); }
  };
#endif

portMUX_TYPE mux = portMUX_INITIALIZER_UNLOCKED;
volatile uint32_t coinPulses = 0, coinLastMs = 0, billPulses = 0, billLastMs = 0;
uint32_t testBtnLastMs = 0;

void IRAM_ATTR onCoinPulse() {
  uint32_t now = millis();
  portENTER_CRITICAL_ISR(&mux);
  if (now - coinLastMs >= PULSE_DEBOUNCE_MS) { coinPulses++; coinLastMs = now; }
  portEXIT_CRITICAL_ISR(&mux);
}
void IRAM_ATTR onBillPulse() {
  uint32_t now = millis();
  portENTER_CRITICAL_ISR(&mux);
  if (now - billLastMs >= PULSE_DEBOUNCE_MS) { billPulses++; billLastMs = now; }
  portEXIT_CRITICAL_ISR(&mux);
}
void sendCredit(uint32_t pesos) {
  if (pesos == 0 || pesos > 5000) return;
  char msg[16]; snprintf(msg, sizeof(msg), "c%lu\n", (unsigned long)pesos);
  Serial.print(msg);
#if BT_MODE == 1
  if (bleKeyboard.isConnected()) {
    char num[12]; snprintf(num, sizeof(num), "c%lu", (unsigned long)pesos);
    bleKeyboard.print(num); bleKeyboard.write(KEY_RETURN);
  }
#elif BT_MODE == 2
  if (bleClientConnected && pTxChar != nullptr) {
    pTxChar->setValue((uint8_t *)msg, strlen(msg)); pTxChar->notify();
  }
#endif
  digitalWrite(LED_PIN, HIGH); delay(50); digitalWrite(LED_PIN, LOW);
}
void setup() {
  Serial.begin(115200);
  pinMode(LED_PIN, OUTPUT);
  pinMode(COIN_PIN, INPUT_PULLUP);
  pinMode(BILL_PIN, INPUT_PULLUP);
  pinMode(TEST_BTN_PIN, INPUT_PULLUP);
  attachInterrupt(digitalPinToInterrupt(COIN_PIN), onCoinPulse, FALLING);
  attachInterrupt(digitalPinToInterrupt(BILL_PIN), onBillPulse, FALLING);
#if BT_MODE == 1
  bleKeyboard.begin();
#elif BT_MODE == 2
  BLEDevice::init(DEVICE_NAME);
  BLEServer *srv = BLEDevice::createServer(); srv->setCallbacks(new NusCallbacks());
  BLEService *svc = srv->createService(NUS_SERVICE_UUID);
  pTxChar = svc->createCharacteristic(NUS_TX_UUID, BLECharacteristic::PROPERTY_NOTIFY);
  pTxChar->addDescriptor(new BLE2902());
  svc->start();
  BLEAdvertising *adv = BLEDevice::getAdvertising();
  adv->addServiceUUID(NUS_SERVICE_UUID); adv->setScanResponse(true);
  BLEDevice::startAdvertising();
#endif
}
void loop() {
  uint32_t now = millis(), c = 0, b = 0;
  portENTER_CRITICAL(&mux);
  if (coinPulses > 0 && (now - coinLastMs) >= COIN_TRAIN_GAP_MS) { c = coinPulses; coinPulses = 0; }
  if (billPulses > 0 && (now - billLastMs) >= BILL_TRAIN_GAP_MS) { b = billPulses; billPulses = 0; }
  portEXIT_CRITICAL(&mux);
  if (c > 0) sendCredit(c * COIN_PESOS_PER_PULSE);
  if (b > 0) sendCredit(b * BILL_PESOS_PER_PULSE);
  if (digitalRead(TEST_BTN_PIN) == LOW && (now - testBtnLastMs) > 600) { testBtnLastMs = now; sendCredit(10); }
  delay(10);
}

Turn it on in photobooth4u

  1. Open Setup → 7 Session → Guest payment.
  2. Switch on Cash acceptor (coins / bills, ESP32).
  3. Pick how the ESP32 connects (default: Bluetooth or USB keyboard).
  4. Set your Price per session (or turn on Multiple payment options).
  5. Tap Test: add ₱10 (or press the ESP32 BOOT button) to confirm credits count up before switching to Photo Session.

2. PayMongo QR Ph: Default Server vs Your Own Cloudflare Account

How PayMongo works by default

Out of the box, you do not need any third-party site or Cloudflare account. Connect your PayMongo secret key once in My Events → Guest payments:

  • photobooth4u's Netlify backend talks directly to api.paymongo.com and automatically sets up the PayMongo webhook on your account.
  • Your secret key is encrypted at rest (AES-256-GCM), write-only, and never sent to the booth tablet.
  • 100% of guest payments go straight to your own PayMongo account (minus PayMongo's standard 1.5% QR Ph fee).

Optional: Host PayMongo API & Webhook on YOUR OWN Cloudflare Account

If you want full zero-trust control where your PayMongo secret key and webhook signing secret never leave your own Cloudflare account, deploy the bundled Worker in tools/cloudflare-paymongo-worker/:

  1. Install Wrangler and deploy to your Cloudflare account: 
    cd tools/cloudflare-paymongo-worker
npx wrangler login
npx wrangler deploy
  2. Set your PayMongo secret key and an optional booth access token inside Cloudflare: 
    npx wrangler secret put PAYMONGO_SECRET_KEY
npx wrangler secret put BOOTH_TOKEN
  3. (Optional — for instant webhook confirmation) Create a KV namespace (npx wrangler kv namespace create PAY_KV), bind it in wrangler.toml, register https://<your-worker>.workers.dev/webhook in PayMongo for payment.paid, and run npx wrangler secret put PAYMONGO_WEBHOOK_SECRET. Even without KV, the Worker polls PayMongo directly every 2.5 seconds.
  4. In photobooth4u → Setup → 7 Session → Guest payment → expand Self-hosted PayMongo (your own Cloudflare Worker), paste your Worker URL + optional token, tap Test connection, then Save Worker.

Bonus: because your booth talks directly to your Cloudflare Worker over HTTPS, QR payment also works inside the Android and Windows/Mac Offline Apps whenever Wi-Fi is available.
